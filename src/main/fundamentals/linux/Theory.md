Команды:

- pwd - где я
- ls - список файлов в текущем каталоге
- ls -ls - подробно список файлов в текущем каталоге (права, размер, дата)
- cd {папка} - зайти в папку
- cd .. - выйти на уровень выше
- cd ~
- cat - вывести весь файл
- head -5 - вывести первые 5 строк

- grep "текст" файл - найти строки с текстом
- cut
- cut -d: -f1
- sort
- grep -v '^#'
- awk '{print $2, $1}'
- sort -nr
- bash script.sh  - запустить скрипт
- echo "текст" - вывести текст
- ptintf
- ${#var} - длина строки в переменной var
- chmod
- chmod +x
- shellchek
- sudo
- find
- case
- tar
- sed
- file
